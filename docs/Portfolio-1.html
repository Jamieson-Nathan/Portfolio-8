# Create "outliers" 
outliers <- watchlist %>%
  mutate(Difference = `Your Rating` - `IMDb Rating`)

top_outliers <- outliers %>% 
  slice_max(abs(Difference), n = 30)  # Get top 30 biggest differences to look orderly

ggplot(top_outliers, aes(x = reorder(Title, Difference), y = Difference, fill = Difference > 0)) +
  geom_bar(stat = "identity") +
  coord_flip() +
  labs(title = "Top 30 Movies We Disagree On",
       x = "Title",
       y = "Difference (Your Rating - IMDb Rating)") +
  scale_fill_manual(values = c("red", "blue"), labels = c("Lower Than IMDb", "Higher Than IMDb")) +
  theme_minimal() +
  theme(text = element_text(size = 16),
        axis.text.y = element_text(size = 8),
        plot.title = element_text(size = 20, face = "bold"))
